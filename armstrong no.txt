#include <stdio.h>
#include<math.h>
void main() {
    int x,y,s=0,r,p;
    printf("enter the no\n");
    scanf("%d",&x);
    p=x;
    while(x>0)
    {
        y=x%10;
        r=pow(y,3);
        s=r+s;
        x=x/10;
        
    }printf("\nthe sum of digits=%d\n",s);
    if(p==s)
    printf ("armstrong no\n");
    else
    printf("not armstrong\n");
}
